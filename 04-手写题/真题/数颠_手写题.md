// 并发 5 次
const requests = [1,2,4,3,3,3];
const count = 5;
let idx = 0;

function request() {
  const re= requests.pop(); // 修笔误：request→requests
  requests.unshift(re);
  return new Promise(resolve => setTimeout(() => resolve(re), 1000));
}

async function start() {
  while(true) {
    const promiseList = [];
    for( let i = 0; i < count; i++) {
      promiseList.push(request());
    }
    await Promise.all(promiseList);
  }
}
start();


// 红绿灯
const a = [['红', 5], ['lv', 2], ['deng', 1]];
function start() {
  const [color, time] = a.shift();
  setTimeout(() => {
    console.log(color);
    start()
  }, time * 1000);
}
